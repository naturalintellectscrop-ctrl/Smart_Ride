    53→// Error Boundary for Map Component
    54→class MapErrorBoundary extends Component<{ children: ReactNode; fallback: ReactNode }, { hasError: boolean }> {
    55→  state = { hasError: false };
    56→
    57→  static getDerivedStateFromError() {
    58→    return { hasError: true };
    59→  }
    60→
    61→  render() {
    62→    if (this.state.hasError) {
    63→      return this.props.fallback;
    64→    }
    65→    return this.props.children;
    66→  }
    67→}
    68→
    69→// Safe Map Component with fallback
    70→function SafeMapView(props: MapViewProps) {
    71→  const fallback = (
    72→    <View style={styles.mapFallback}>
    73→      <Text style={styles.mapFallbackEmoji}>🗺️</Text>
    74→      <Text style={styles.mapFallbackText}>Map unavailable</Text>
    75→      <Text style={styles.mapFallbackSubtext}>
    76→        Location: {props.initialRegion?.latitude?.toFixed(4)}, {props.initialRegion?.longitude?.toFixed(4)}
    77→      </Text>
    78→    </View>
    79→  );
    80→
    81→  return (
    82→    <MapErrorBoundary fallback={fallback}>
    83→      <MapView {...props} />
    84→    </MapErrorBoundary>
    85→  );
    86→}

   629→                </GlassCard>
   630→              </Animated.View>
   631→            )}
   632→
   633→            {/* Go Online / Go Offline Button */}
   634→            <View style={styles.toggleButtonContainer}>

   636→                title={isOnline ? 'Go Offline' : 'Go Online'}

   654→// Pulsing Loader Component
   655→function PulsingLoader() {
   656→  const scale = useSharedValue(1);
   657→
   658→  useEffect(() => {
   659→    scale.value = withRepeat(
   660→      withSequence(
   661→        withTiming(1.2, { duration: 600 }),
   662→        withTiming(1, { duration: 600 })
   663→      ),
   664→      -1,
   665→      true
   666→    );
   667→  }, []);
   668→
   669→  const animatedStyle = useAnimatedStyle(() => ({
   670→    transform: [{ scale: scale.value }],
   671→  }));
   672→
   673→  return (
   674→    <Animated.View style={animatedStyle}>
   675→      <ActivityIndicator size="large" color={COLORS.primary} />
   676→    </Animated.View>
   677→  );
   678→}

   690→// Animated Pressable Wrapper (for button press scale effect)
   691→function AnimatedPressable({ children, onPress, disabled }: { children: React.ReactNode; onPress: () => void; disabled?: boolean }) {
   692→  const scale = useSharedValue(1);
   693→
   694→  const handlePressIn = () => {
   695→    if (!disabled) {
   696→      scale.value = withSpring(0.96, { damping: 15, stiffness: 300 });
   697→    }
   698→  };
   699→
   700→  const handlePressOut = () => {
   701→    scale.value = withSpring(1, { damping: 15, stiffness: 300 });
   702→  };
   703→
   704→  const animatedStyle = useAnimatedStyle(() => ({
   705→    transform: [{ scale: scale.value }],
   706→  }));
   707→
   708→  return (
   709→    <TouchableOpacity
   710→      onPress={onPress}
   711→      onPressIn={handlePressIn}
   712→      onPressOut={handlePressOut}
   713→      disabled={disabled}
   714→      activeOpacity={0.95}
   715→      style={{ flex: 1 }}
   716→    >
   717→      <Animated.View style={animatedStyle}>
   718→        {children}
   719→      </Animated.View>
   720→    </TouchableOpacity>
   721→  );
   722→}
